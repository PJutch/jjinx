Jjinx

RIIR nginx clone

Features

  • Static websites
  • Reverse proxy
  • Обнаружение сбоев
  • Поддержа HTTP/1.1 и HTTPS

Мин. конфиг

server {
    port 8080
    default

    route = / {
        file ./
    }
}

server {
    port 4433
    default

    ssl_cert server.crt
    ssl_keys server.key

    route / {
        proxy http://localhost:8080
    }
}

Реверс прокси

upstream backend {
    round_robin
    server app1.example.com
    server app2.example.com

    max_fails 2
    connect_timeout 10
}

server {
    port 8080
    default

    route /proxy/ {
        proxy http://backend:8000/
    }
}
